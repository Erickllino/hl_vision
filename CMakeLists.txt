cmake_minimum_required(VERSION 3.16)
project(hl_vision_preprocess LANGUAGES CXX)

set(HSL_PLAYER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../hsl-player" CACHE PATH "hsl-player checkout containing the shared OpenCV-only preprocessor")
set(VISION_ROOT "${HSL_PLAYER_ROOT}/src/vision")
foreach(FILE include/booster_vision/model/line_preprocessor.h src/model/line_preprocessor.cpp)
  if(NOT EXISTS "${VISION_ROOT}/${FILE}")
    message(FATAL_ERROR "Missing shared source: ${VISION_ROOT}/${FILE}. Set HSL_PLAYER_ROOT to the updated hsl-player checkout.")
  endif()
endforeach()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(shared_line_preprocessor STATIC "${VISION_ROOT}/src/model/line_preprocessor.cpp")
target_include_directories(shared_line_preprocessor PUBLIC "${VISION_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(shared_line_preprocessor PUBLIC ${OpenCV_LIBS})
target_compile_features(shared_line_preprocessor PUBLIC cxx_std_17)
add_executable(hl_preprocess tools/preprocess.cpp)
target_link_libraries(hl_preprocess PRIVATE shared_line_preprocessor)
